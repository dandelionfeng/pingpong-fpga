cmake_minimum_required(VERSION 3.20)
project(AFH03FieldTest LANGUAGES CXX)
set(PIPELINE_PACKAGE_SOURCE "" CACHE PATH "Selected runtime source directory")
set(RKNN_INCLUDE_DIR "" CACHE PATH "Matching rknn_api.h directory")
set(RKNN_RUNTIME_LIBRARY "/usr/lib/librknnrt.so" CACHE FILEPATH "Board RKNN runtime")
if(NOT EXISTS "${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline/CMakeLists.txt")
  message(FATAL_ERROR "PIPELINE_PACKAGE_SOURCE must contain the selected source")
endif()
set(PIPELINE_B_MODULE_DIR "${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline/modules/b_measurement" CACHE PATH "" FORCE)
set(PIPELINE_C_MODULE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/c_rknn" CACHE PATH "" FORCE)
add_subdirectory("${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline" pipeline)
add_executable(motion7_chain_bench motion7_chain_bench.cpp)

target_include_directories(motion7_chain_bench PRIVATE "${PIPELINE_B_MODULE_DIR}/src" "${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline/trajectory/include")
target_link_libraries(motion7_chain_bench PRIVATE pipeline_core trajectory)
add_executable(mask_release_cpu_bench mask_release_cpu_bench.cpp)
target_include_directories(mask_release_cpu_bench PRIVATE "${PIPELINE_B_MODULE_DIR}/src" "${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline/trajectory/include")
target_link_libraries(mask_release_cpu_bench PRIVATE pipeline_core trajectory)
