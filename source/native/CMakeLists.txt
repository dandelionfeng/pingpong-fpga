cmake_minimum_required(VERSION 3.20)
project(StereoServeRuntime LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(PIPELINE_PACKAGE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/base" CACHE PATH "Runtime base source")
set(RKNN_INCLUDE_DIR "" CACHE PATH "Matching rknn_api.h directory")
set(RKNN_RUNTIME_LIBRARY "/usr/lib/librknnrt.so" CACHE FILEPATH "Board RKNN runtime")
set(PIPELINE_B_MODULE_DIR "${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline/modules/b_measurement" CACHE PATH "" FORCE)
set(PIPELINE_C_MODULE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/rk_runtime/c_rknn" CACHE PATH "" FORCE)
add_subdirectory("${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline" pipeline)
add_executable(motion7_chain_kinematic optimized/motion7_chain_bench.cpp)
target_include_directories(motion7_chain_kinematic PRIVATE optimized rk_runtime rk_runtime/c_rknn
 "${PIPELINE_B_MODULE_DIR}/src" "${PIPELINE_PACKAGE_SOURCE}/software/pc_pipeline/trajectory/include")
target_link_libraries(motion7_chain_kinematic PRIVATE pipeline_core trajectory)
