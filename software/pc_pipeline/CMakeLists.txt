cmake_minimum_required(VERSION 3.20)
project(AUsbStereoPipeline VERSION 1.0.0 LANGUAGES CXX)
# This MSVC installation emits UTF-8 Chinese, while CMake's probe decodes it
# with the system code page. Correct the probe only when that mismatch occurs.
if(MSVC AND CMAKE_CXX_CL_SHOWINCLUDES_PREFIX MATCHES "娉ㄦ剰")
  set(CMAKE_CL_SHOWINCLUDES_PREFIX "注意: 包含文件:  ")
endif()
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(UNIX)
  set(CMAKE_INSTALL_RPATH "$ORIGIN/../lib")
endif()
option(PIPELINE_BUILD_GUI "Build Qt replay interface" ON)
option(PIPELINE_SANITIZE "Enable address/undefined sanitizers on Linux" OFF)
set(PIPELINE_B_MODULE_DIR "" CACHE PATH "B module with register_b_module(Registry&) entry")
set(PIPELINE_C_MODULE_DIR "" CACHE PATH "C module with register_c_module(Registry&) entry")
set(PIPELINE_DMA_MODULE_DIR "" CACHE PATH "Vendor transport with register_dma_module(Registry&) entry")
include(cmake/LocalDependencies.cmake)
find_package(OpenCV 4.13 EXACT REQUIRED COMPONENTS core imgproc imgcodecs)
find_package(Threads REQUIRED)
include(FetchContent)
FetchContent_Declare(json URL https://github.com/nlohmann/json/releases/download/v3.11.3/json.tar.xz URL_HASH SHA256=d6c65aca6b1ed68e7a182f4757257b107ae403032760ed6ef121c9d55e81757d DOWNLOAD_EXTRACT_TIMESTAMP TRUE)
FetchContent_MakeAvailable(json)
add_library(pipeline_module_api INTERFACE)
target_include_directories(pipeline_module_api INTERFACE include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pipeline_module_api INTERFACE ${OpenCV_LIBS} Threads::Threads nlohmann_json::nlohmann_json)
target_compile_features(pipeline_module_api INTERFACE cxx_std_17)
add_library(pipeline_core src/io.cpp src/motion.cpp src/backend.cpp src/runner.cpp src/v4l2.cpp src/sha256.cpp src/pcie.cpp)
target_include_directories(pipeline_core PUBLIC include ${OpenCV_INCLUDE_DIRS})
target_link_libraries(pipeline_core PUBLIC ${OpenCV_LIBS} Threads::Threads nlohmann_json::nlohmann_json)
target_compile_definitions(pipeline_core PUBLIC PIPELINE_VERSION="1.0.0")
if(MSVC)
  target_compile_options(pipeline_core PUBLIC /utf-8 /W4)
  target_compile_options(pipeline_module_api INTERFACE /utf-8 /W4)
else()
  target_compile_options(pipeline_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(PIPELINE_SANITIZE AND NOT MSVC)
  target_compile_options(pipeline_core PUBLIC -fsanitize=address,undefined -fno-omit-frame-pointer)
  target_link_options(pipeline_core PUBLIC -fsanitize=address,undefined)
endif()
foreach(role B C DMA)
  if(PIPELINE_${role}_MODULE_DIR)
    add_subdirectory("${PIPELINE_${role}_MODULE_DIR}" "module_${role}")
    string(TOLOWER "${role}" lower)
    target_link_libraries(pipeline_core PUBLIC pipeline_${lower}_module)
    target_compile_definitions(pipeline_core PRIVATE PIPELINE_HAS_${role}_MODULE=1)
  endif()
endforeach()
add_executable(pipeline_cli src/cli.cpp)
target_link_libraries(pipeline_cli PRIVATE pipeline_core)
include(CTest)
if(BUILD_TESTING)
  add_executable(pipeline_tests tests/core_tests.cpp)
  target_link_libraries(pipeline_tests PRIVATE pipeline_core)
  add_test(NAME core_contracts COMMAND pipeline_tests)
endif()
if(PIPELINE_BUILD_GUI)
  find_package(Qt6 6.8.3 EXACT REQUIRED COMPONENTS Widgets OpenGL Test)
  set(CMAKE_AUTOMOC ON)
  add_executable(pipeline_gui WIN32 src/gui.cpp)
  target_link_libraries(pipeline_gui PRIVATE pipeline_core Qt6::Widgets Qt6::OpenGL Qt6::Test)
endif()
# Deploy the selected Debug/Release DLLs for builds started directly in an IDE.
# The official Windows OpenCV package exposes its components via opencv_world.
if(WIN32)
  foreach(app pipeline_cli pipeline_gui pipeline_tests)
    if(TARGET ${app})
      foreach(dependency IN LISTS OpenCV_LIBS)
        if(TARGET ${dependency})
          get_target_property(dependency_type ${dependency} TYPE)
          if(dependency_type STREQUAL "SHARED_LIBRARY")
            add_custom_command(TARGET ${app} POST_BUILD
              COMMAND ${CMAKE_COMMAND} -E copy_if_different
                "$<TARGET_FILE:${dependency}>" "$<TARGET_FILE_DIR:${app}>"
              VERBATIM)
          endif()
        endif()
      endforeach()
    endif()
  endforeach()
  if(TARGET pipeline_gui AND TARGET Qt6::windeployqt)
    add_custom_command(TARGET pipeline_gui POST_BUILD
      COMMAND "$<TARGET_FILE:Qt6::windeployqt>"
        "$<IF:$<CONFIG:Debug>,--debug,--release>" --no-translations --no-compiler-runtime
        "$<TARGET_FILE:pipeline_gui>"
      VERBATIM)
  endif()
endif()
install(TARGETS pipeline_cli RUNTIME DESTINATION bin)
if(PIPELINE_BUILD_GUI)
  install(TARGETS pipeline_gui RUNTIME DESTINATION bin)
endif()
